<script>

/* ==========================================================
   NELOY ENTERPRISE
   CABINET CONFIGURATOR ENGINE
========================================================== */


/* STANDARD BOARD */

const STANDARD_BOARD = {

    widthFt: 4,

    heightFt: 8,

    areaSqFt: 32

};


/* SHORTCUT */

function $(id){

    return document.getElementById(id);

}


/* NUMBER INPUT */

function num(id){

    return Math.max(
        0,
        Number($(id).value) || 0
    );

}


/* MONEY FORMAT */

function money(value){

    return "৳" +
        Math.round(value)
        .toLocaleString("en-BD");

}


/* ==========================================================
   MAIN CALCULATION
========================================================== */

function calculate(){


    /* -------------------------
       DIMENSIONS
    ------------------------- */

    const H = num("height");

    const W = num("width");

    const D = num("depth");


    /* -------------------------
       COMPARTMENTS
    ------------------------- */

    const C =
        Math.max(
            1,
            Math.round(
                num("compartments")
            )
        );


    const shelvesPer =
        Math.max(
            0,
            Math.round(
                num("shelves")
            )
        );


    /* -------------------------
       DOOR
    ------------------------- */

    const doorType =
        $("doorType").value;


    const tracks =
        Math.max(
            0,
            Math.round(
                num("slidingSections")
            )
        );


    const clearanceIn =
        num("clearance");


    const glassDoors =
        Math.max(
            0,
            Math.round(
                num("glassDoors")
            )
        );


    /* -------------------------
       PRICES
    ------------------------- */

    const boardPrice =
        num("boardPrice");


    const fittingPrice =
        num("fittingPrice");


    const glassPrice =
        num("glassPrice");


    const ePrice =
        num("ePrice");


    const hardwarePrice =
        num("hardwarePrice");


    const backBoards =
        Math.max(
            0,
            Math.round(
                num("backBoards")
            )
        );


    /* ======================================================
       BASIC CARCASS CUTTING LIST
    ====================================================== */


    /*
       2 outside sides
    */

    const sideQty = 2;


    /*
       4 compartments require
       3 internal partitions
    */

    const partitionQty =
        Math.max(
            0,
            C - 1
        );


    /*
       Each compartment width
    */

    const compartmentWidth =
        W / C;


    /*
       Shelf quantity
    */

    const shelfQty =
        C * shelvesPer;


    /*
       PIECES
    */

    const pieces = [


        {

            name:
            "Left + Right Side",

            qty:
            sideQty,

            width:
            D,

            height:
            H,

            area:
            sideQty * D * H

        },


        {

            name:
            "Internal Partitions",

            qty:
            partitionQty,

            width:
            D,

            height:
            H,

            area:
            partitionQty * D * H

        },


        {

            name:
            "Top + Bottom",

            qty:
            2,

            width:
            D,

            height:
            W,

            area:
            2 * D * W

        },


        {

            name:
            "Shelves",

            qty:
            shelfQty,

            width:
            D,

            height:
            compartmentWidth,

            area:
            shelfQty *
            D *
            compartmentWidth

        }

    ];


    /* TOTAL CARCASS AREA */

    const carcassArea =
        pieces.reduce(
            function(sum,p){

                return sum + p.area;

            },
            0
        );


    /* ======================================================
       BOARD QUANTITY
    ====================================================== */


    const extraBackBoardArea =
        backBoards *
        STANDARD_BOARD.areaSqFt;


    const totalBoardArea =
        carcassArea +
        extraBackBoardArea;


    const boardQty =
        Math.ceil(
            totalBoardArea /
            STANDARD_BOARD.areaSqFt
        );


    /* ======================================================
       GLASS DOOR CALCULATION
    ====================================================== */


    let doorQty = 0;

    let glassArea = 0;

    let doorWidth = 0;


    if(
        doorType ===
        "glassSliding"
    ){

        doorQty =
            glassDoors > 0
            ? glassDoors
            : Math.max(2,tracks);


        /*
           2-inch clearance converted
           from inch to feet.
        */

        const clearanceFt =
            clearanceIn / 12;


        /*
           Usable front width
        */

        const usableWidth =
            Math.max(
                0,
                W - clearanceFt
            );


        /*
           Width of each glass door
        */

        doorWidth =
            usableWidth /
            doorQty;


        /*
           Total glass
        */

        glassArea =
            doorQty *
            doorWidth *
            H;

    }


    else if(
        doorType ===
        "glassHinged"
    ){

        doorQty =
            glassDoors > 0
            ? glassDoors
            : C;


        doorWidth =
            W / doorQty;


        glassArea =
            doorQty *
            doorWidth *
            H;

    }


    /* ======================================================
       E-CHANNEL
    ====================================================== */


    let eLength = 0;


    if(
        doorType ===
        "glassSliding" &&
        tracks > 0
    ){

        /*
           E-channel =
           cabinet width × tracks
        */

        eLength =
            W * tracks;

    }


    /* ======================================================
       COST CALCULATION
    ====================================================== */


    const boardCost =
        boardQty *
        boardPrice;


    const fittingCost =
        boardQty *
        fittingPrice;


    const glassCost =
        glassArea *
        glassPrice;


    const eCost =
        eLength *
        ePrice;


    const hardwareCost =
        doorQty *
        hardwarePrice;


    const total =
        boardCost +
        fittingCost +
        glassCost +
        eCost +
        hardwareCost;


    /* ======================================================
       DISPLAY TOTAL
    ====================================================== */


    $("configuredTotal")
        .textContent =
        money(total);


    /* ======================================================
       MATERIAL BREAKDOWN
    ====================================================== */


    const rows = [


        {

            name:"Board",

            detail:
            boardQty +
            " board × " +
            money(boardPrice),

            cost:
            boardCost

        },


        {

            name:"Fitting",

            detail:
            boardQty +
            " board × " +
            money(fittingPrice),

            cost:
            fittingCost

        },


        {

            name:"Glass",

            detail:
            glassArea.toFixed(2) +
            " sq ft × " +
            money(glassPrice),

            cost:
            glassCost

        },


        {

            name:"E-channel",

            detail:
            eLength.toFixed(2) +
            " ft × " +
            money(ePrice),

            cost:
            eCost

        },


        {

            name:"Door Hardware",

            detail:
            doorQty +
            " × " +
            money(hardwarePrice),

            cost:
            hardwareCost

        }

    ];


    $("breakdown").innerHTML =

        rows.map(

            function(row){

                let displayCost;

                if(
                    row.cost > 0
                ){

                    displayCost =
                        money(row.cost);

                }

                else{

                    displayCost =
                        '<span class="warning">' +
                        'Not configured' +
                        '</span>';

                }


                return `

                <tr>

                    <td>

                        <strong>
                        ${row.name}
                        </strong>

                        <br>

                        <span class="muted">
                        ${row.detail}
                        </span>

                    </td>

                    <td>
                        ${displayCost}
                    </td>

                </tr>

                `;

            }

        ).join("");


    /* ======================================================
       CUTTING LIST
    ====================================================== */


    $("cuttingList").innerHTML =

        pieces.map(

            function(piece){

                return `

                <tr>

                    <td>

                        <strong>
                        ${piece.name}
                        </strong>

                    </td>

                    <td>

                        ${piece.qty}
                        ×
                        ${piece.width.toFixed(2)}
                        ft
                        ×
                        ${piece.height.toFixed(2)}
                        ft

                        <br>

                        <span class="muted">

                        ${piece.area.toFixed(2)}
                        sq ft

                        </span>

                    </td>

                </tr>

                `;

            }

        ).join("");


    /* ======================================================
       BOARD DETAILS
    ====================================================== */


    const unusedArea =

        Math.max(

            0,

            boardQty *
            STANDARD_BOARD.areaSqFt
            -
            totalBoardArea

        );


    let details = `

        <p>

        <strong>
        Cabinet:
        </strong>

        ${H} ft ×
        ${W} ft ×
        ${D} ft

        </p>


        <p>

        <strong>
        Compartments:
        </strong>

        ${C}

        </p>


        <p>

        <strong>
        Internal partitions:
        </strong>

        ${partitionQty}

        </p>


        <p>

        <strong>
        Shelves:
        </strong>

        ${shelfQty}

        </p>


        <p>

        <strong>
        Total cutting area:
        </strong>

        ${carcassArea.toFixed(2)}
        sq ft

        </p>


        <p>

        <strong>
        Standard board:
        </strong>

        8 × 4 ft =
        32 sq ft

        </p>


        <p>

        <strong>
        Board estimate:
        </strong>

        ${boardQty} board(s)

        </p>


        <p>

        <strong>
        Remaining area before
        cutting loss:
        </strong>

        ${unusedArea.toFixed(2)}
        sq ft

        </p>

    `;


    /* GLASS INFORMATION */

    if(
        doorType ===
        "glassSliding"
    ){

        const clearanceFt =
            clearanceIn / 12;


        details += `

        <p>

        <strong>
        Glass sliding door:
        </strong>

        ${doorQty} doors

        </p>


        <p>

        <strong>
        Door clearance:
        </strong>

        ${clearanceIn}
        inch

        </p>


        <p>

        <strong>
        Usable front width:
        </strong>

        ${Math.max(
            0,
            W - clearanceFt
        ).toFixed(2)}
        ft

        </p>


        <p>

        <strong>
        Approx. each door width:
        </strong>

        ${doorWidth.toFixed(2)}
        ft

        </p>


        <p>

        <strong>
        Glass required:
        </strong>

        ${glassArea.toFixed(2)}
        sq ft

        </p>


        <p>

        <strong>
        E-channel:
        </strong>

        ${eLength.toFixed(2)}
        ft

        </p>

        `;

    }


    if(
        glassArea > 0 &&
        glassPrice === 0
    ){

        details += `

        <p class="warning">

        <strong>
        Glass price not configured.
        </strong>

        Quantity is calculated,
        but no confirmed glass price
        has been entered.

        </p>

        `;

    }


    if(
        eLength > 0 &&
        ePrice === 0
    ){

        details += `

        <p class="warning">

        <strong>
        E-channel price not configured.
        </strong>

        Required:
        ${eLength.toFixed(2)}
        ft

        </p>

        `;

    }


    $("boardDetails")
        .innerHTML =
        details;


    /* ======================================================
       PREVIEW
    ====================================================== */

    renderPreview(
        C,
        shelvesPer,
        doorType
    );


    /* STATUS */

    $("statusText").innerHTML =

        `

        Board:
        <strong>
        ${boardQty}
        </strong>

        ·

        Fitting:
        <strong>
        ${money(fittingCost)}
        </strong>

        `;

}


/* ==========================================================
   CABINET PREVIEW
========================================================== */

function renderPreview(
    compartmentCount,
    shelvesPer,
    doorType
){

    const preview =
        $("cabinetPreview");


    preview.innerHTML = "";


    for(
        let i=0;
        i<compartmentCount;
        i++
    ){

        const compartment =
            document.createElement(
                "div"
            );


        compartment.className =
            "compartment";


        /* SHELVES */

        for(
            let s=1;
            s<=shelvesPer;
            s++
        ){

            const shelf =
                document.createElement(
                    "div"
                );


            shelf.className =
                "shelf";


            shelf.style.top =
                (
                    s /
                    (shelvesPer + 1)
                    *
                    100
                ) + "%";


            compartment.appendChild(
                shelf
            );

        }


        /* GLASS */

        if(
            doorType ===
            "glassSliding" ||

            doorType ===
            "glassHinged"
        ){

            const door =
                document.createElement(
                    "div"
                );


            door.className =
                "glass-door";


            door.innerHTML =
                "<span>GLASS</span>";


            compartment.appendChild(
                door
            );

        }


        preview.appendChild(
            compartment
        );

    }

}
